def calculate_overnight_sizing(consensus: str, whisper: str, profile: str, pattern: str, narrative: str) -> str:
    """
    Calculates the binary leveraged overnight risk allocation 
    based on a 5-stage left-to-right structural precedence pipeline.
    
    CRITICAL RISK FIX: 'Bearish' narrative is an absolute kill-switch (ZERO).
    """
    # Stage 1 & 2: Apex Fundamental & Media Narrative Gatekeepers
    if consensus == "Negative" or whisper == "Miss" or narrative == "Bearish":
        return "ZERO"
        
    # Stage 3: Asset Profile Behavior Filter
    if profile in ["Value", "Cyclical", "Defensive"]:
        return "ZERO"
        
    # Stage 4: Technical Pattern Structural Filter
    if pattern in ["Neutral", "Bearish"]:
        return "ZERO"
        
    # Stage 5: Core Execution Matrix (Growth + Beat + Valid Pattern + Non-Bearish Narrative)
    if pattern in ["Breakout", "Confirmed"]:
        if narrative in ["Bullish", "Conflicting"]:
            return "PRIORITY"
        elif narrative == "Cautious":
            return "STANDARD"
            
    elif pattern == "Coiling":
        if narrative == "Bullish":
            return "STANDARD"
        else:
            return "ZERO" # Unconfirmed coiling under any friction (Cautious/Conflicting) is a skip
            
    return "ZERO" # Catch-all safety fallback
